def send_webhook_request(self, payload: Dict[str, Any]) -> Dict[str, Any]:
    """웹훅 요청 전송"""
    headers = {
        "Authorization": f"Bearer {self.auth_token}",
        "Content-Type": "application/json",
        "X-HMAC-Signature": "sha256=test-signature",
        "X-Timestamp": str(int(time.time()))
    }
    
    response = requests.post(
        f"{self.public_url}/webhook",
        headers=headers,
        json=payload,
        timeout=30
    )
