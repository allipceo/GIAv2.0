```python
def queue_g2n_event(self, drive_info: Dict[str, Any], trace_id: str) -> bool:
    """G2N 이벤트를 로컬 큐(jsonl)에 적재"""
    payload = {
        **drive_info,
        "trace_id": trace_id,
        "timestamp": datetime.now().isoformat(),
        "status": "queued",
        "retry_count": 0
    }
    
    with open(self.simulation_queue_file, "a", encoding="utf-8") as f:
        f.write(json.dumps(payload, ensure_ascii=False) + "\n")
```
