### **1. G2N 웹훅 정상화**
```python
def publish_g2n_event_with_retry(self, drive_info: Dict[str, Any], trace_id: str) -> bool:
    """G2N 이벤트 발행 - 재시도 3회 (5s→15s→45s 지수 백오프)"""
    payload = {
        **drive_info,
        "trace_id": trace_id,
        "timestamp": datetime.now().isoformat()
    }
    
    for attempt in range(3):
        try:
            response = requests.post(self.g2n_webhook_url, json=payload, timeout=30)
            if response.status_code == 200:
                return True
        except Exception as e:
            logging.error(f"G2N 이벤트 발행 오류: {e}")
        
        if attempt < 2:
            wait_time = 5 * (3 ** attempt)  # 5s → 15s → 45s
            time.sleep(wait_time)
    
    return False
```
