```python
def check_g2n_health(self) -> Dict[str, Any]:
    """G2N 웹훅 헬스 체크"""
    try:
        response = requests.get('http://localhost:8000/health', timeout=10)
        if response.status_code == 200:
            return {"status": "success", "health_data": response.json()}
        else:
            return {"status": "error", "status_code": response.status_code}
    except Exception as e:
        return {"status": "error", "error": str(e)}

def check_notion_me(self) -> Dict[str, Any]:
    """Notion me API 체크"""
    try:
        headers = {"Authorization": f"Bearer {self.notion_token}"}
        response = requests.get("https://api.notion.com/v1/users/me", headers=headers)
        if response.status_code == 200:
            return {"status": "success", "me_data": response.json()}
        else:
            return {"status": "error", "status_code": response.status_code}
    except Exception as e:
        return {"status": "error", "error": str(e)}
```
