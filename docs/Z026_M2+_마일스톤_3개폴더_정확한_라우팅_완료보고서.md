```python
def get_target_db_id(self, folder_name: str) -> Optional[str]:
    """
    M2+ 마일스톤: 3개 폴더 구조에 따른 정확한 라우팅
    """
    if folder_name == "01_개발문서":
        target_db_id = os.getenv('ZOBIS_DEV_DB_ID')
        logger.info(f"라우팅: {folder_name} -> ZOBIS 개발문서 DB")
    elif folder_name == "02_프로젝트문서":
        target_db_id = os.getenv('PROJECT_RESOURCES_DB_ID')
        logger.info(f"라우팅: {folder_name} -> 프로젝트리소스 DB")
    elif folder_name == "03_기타문서":
        target_db_id = os.getenv('RESOURCE_HUB_DB_ID')
        logger.info(f"라우팅: {folder_name} -> Resource Hub DB")
    else:
        # INBOX 루트에 직접 업로드된 경우 기본값
        target_db_id = os.getenv('ZOBIS_DEV_DB_ID')
        logger.warning(f"알 수 없는 폴더: {folder_name} -> 기본값 (ZOBIS 개발문서 DB)")
    
    return target_db_id
```

### **3단계: 폴더 내부 파일 처리 로직 수정**

#### **수정된 `process_files` 함수**
```python
def process_files(self, file_limit: int = None) -> Dict:
    """
    M2+ 마일스톤: 3개 하위 폴더의 파일들을 각각 처리합니다.
    """
    # 하위 폴더 ID들
    subfolders = {
        "01_개발문서": os.getenv('GOOGLE_DRIVE_DEV_FOLDER_ID'),
        "02_프로젝트문서": os.getenv('GOOGLE_DRIVE_PROJECT_FOLDER_ID'),
        "03_기타문서": os.getenv('GOOGLE_DRIVE_OTHER_FOLDER_ID')
    }
    
    all_files = []
    
    for folder_name, folder_id in subfolders.items():
        if not folder_id:
            logger.warning(f"폴더 ID가 설정되지 않음: {folder_name}")
            continue
        
        # 각 폴더의 파일들 추출
        folder_files = get_drive_file_metadata(folder_id)
        
        if folder_files:
            logger.info(f"{folder_name}에서 {len(folder_files)}개 파일 발견")
            
            # 각 파일에 폴더 정보 추가
            for file_metadata in folder_files:
                file_metadata['source_folder'] = folder_name
                all_files.append(file_metadata)
        else:
            logger.info(f"{folder_name}에 파일이 없습니다.")
```
